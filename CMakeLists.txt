cmake_minimum_required(VERSION 3.16)
project(TeraTool_UI_Replica LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTOUIC_SEARCH_PATHS ${CMAKE_CURRENT_SOURCE_DIR}/ui)
find_package(Qt5 5.14 REQUIRED COMPONENTS Core Gui Widgets Svg)
add_executable(TeraTool_UI WIN32 main.cpp mainwindow.cpp mainwindow.h shell.h
    theme_manager.cpp theme_manager.h ui/mainwindow.ui artwork.qrc themes.qrc)
target_include_directories(TeraTool_UI PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
if(MSVC)
    target_sources(TeraTool_UI PRIVATE app.manifest)
endif()
include(CTest)
target_link_libraries(TeraTool_UI PRIVATE Qt5::Core Qt5::Gui Qt5::Widgets Qt5::Svg)
if(COMMAND qt5_import_plugins)
    qt5_import_plugins(TeraTool_UI INCLUDE Qt5::QSvgPlugin Qt5::QSvgIconPlugin)
endif()
